4. FIREBASE SETTINGS TO CHECK/UPDATE
================================================================================

── Firestore Security Rules ────────────────────────────────────────────────────

Go to: Firebase Console → Firestore → Rules
Make sure these collections are properly secured:

  rules_version = '2';
  service cloud.firestore {
    match /databases/{database}/documents {
      // Users can read/write their own profile
      match /users/{userId} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
      // Sessions — users can read their own, write their own
      match /sessions/{sessionId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null;
      }
      // Community posts — any authenticated user can read/write
      match /community_posts/{postId} {
        allow read: if true;
        allow write: if request.auth != null;
      }
      // Friend requests
      match /friend_requests/{reqId} {
        allow read, write: if request.auth != null;
      }
      // Bug reports — any authenticated user can create
      match /bug_reports/{reportId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
      }
      // Announcements — anyone can read, only admin can write
      match /announcements/{annoId} {
        allow read: if true;
        allow write: if request.auth != null;
      }
    }
  }

── Firebase Auth Settings ──────────────────────────────────────────────────────

Go to: Firebase Console → Authentication → Settings → Authorized domains
Make sure your Vercel domain is listed:
  - your-app.vercel.app
  - localhost (for development)
